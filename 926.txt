// =========================================================================
// BLOCK 1: COMPACT STATE RETENTION TYPES & STRUCTS
// =========================================================================
typedef struct {
    uint16_t w0; // Maps Word 0: MainWeapon, Fire, LockOn, Eject, etc.
    uint16_t w1; // Maps Word 1: Washing, Extinguisher, Chaff, etc.
    uint16_t w2; // Maps Word 2: Comm5, SightChange, etc.
} compact_channel_state_t;

// Persistent memory structures for independent device input streams
static compact_channel_state_t channel_state_kbm      = {0};
static compact_channel_state_t channel_state_joystick = {0};
static compact_channel_state_t channel_state_a        = {0};
static compact_channel_state_t channel_state_b        = {0};


// =========================================================================
// BLOCK 2: DEVICE-SPECIFIC INPUT PARSING LAYER
// =========================================================================

                                     local_sbc_data_t const* sbc_data, 
                                     uint8_t input_source) 
{
    // ---------------------------------------------------------------------
    // PART A: PARSE AND CACHE INDEPENDENT PERSISTENT BITMASKS
    // ---------------------------------------------------------------------
    if (input_source == INPUT_SOURCE_KBM) {
        channel_state_kbm.w0 = 0;
        channel_state_kbm.w1 = 0;
        channel_state_kbm.w2 = 0;

        if (is_key_pressed(KEY_A))      gp->steel_battalion_in_report.rotationLever = -32768; 
        else if (is_key_pressed(KEY_D)) gp->steel_battalion_in_report.rotationLever = 32767;  
        
        accumulated_aim_x += (local_mouse_x * MOUSE_SENSITIVITY);
        accumulated_aim_y += (local_mouse_y * MOUSE_SENSITIVITY);
        if (accumulated_aim_x > 65535) accumulated_aim_x = 65535; if (accumulated_aim_x < 0) accumulated_aim_x = 0;
        if (accumulated_aim_y > 65535) accumulated_aim_y = 65535; if (accumulated_aim_y < 0) accumulated_aim_y = 0;
        
        gp->steel_battalion_in_report.aimingX = (uint16_t)accumulated_aim_x; 
        gp->steel_battalion_in_report.aimingY = (uint16_t)accumulated_aim_y; 
        
        if (is_key_pressed(KEY_P))     channel_state_kbm.w0 |= (1 << 4);  // CockpitHatch
        if (is_key_pressed(KEY_SPACE)) channel_state_kbm.w0 |= (1 << 3);  // Eject
        if (is_key_pressed(KEY_I))     channel_state_kbm.w0 |= (1 << 5);  // Ignition
        if (is_key_pressed(KEY_ENTER)) channel_state_kbm.w0 |= (1 << 6);  // Start
    }
	
    else if (input_source == INPUT_SOURCE_JOYSTICK) {
        channel_state_joystick.w0 = 0;
        channel_state_joystick.w1 = 0;
        channel_state_joystick.w2 = 0;

        if (pad_data->hat == 0) gp->steel_battalion_in_report.sightChangeY = -8000;
        if (pad_data->hat == 4) gp->steel_battalion_in_report.sightChangeY = 8000;
        
        if (pad_data->buttons & (1 << 0)) channel_state_joystick.w0 |= (1 << 0); // MainWeapon
        if (pad_data->buttons & (1 << 1)) channel_state_joystick.w0 |= (1 << 1); // Fire
        if (pad_data->buttons & (1 << 3)) channel_state_joystick.w0 |= (1 << 2); // LockOn
    } 
    
    else if (input_source == INPUT_SOURCE_A) {
        channel_state_a.w0 = 0;
        if (is_mouse_pressed(MOUSE_BUTTON_LEFT)) {
            channel_state_a.w0 |= (1 << 0); // MainWeapon Map
        }
    }
    
    else if (input_source == INPUT_SOURCE_B) {
        channel_state_b.w0 = 0;
        if (is_mouse_pressed(MOUSE_BUTTON_LEFT)) {
            channel_state_b.w0 |= (1 << 2); // LockOn Map
        }
    }






    // =========================================================================
    // BLOCK 3: PASSTHROUGH & BITWISE AGGREGATION MATRIX
    // =========================================================================
    // ---------------------------------------------------------------------
    // PART B: NATIVE PHYSICAL PASSTHROUGH (SBC handles structural components directly)
    // ---------------------------------------------------------------------

        memset(&gp->steel_battalion_in_report.dButtons, 0, sizeof(gp->steel_battalion_in_report.dButtons));

        uint16_t w1 = sbc_data->buttons[1]; 
        uint16_t w2 = sbc_data->buttons[2];

        gp->steel_battalion_in_report.dButtons.LockOn       = (w0 & (1 << 2)) ? true : false;

        gp->steel_battalion_in_report.dButtons.CockpitHatch = (w0 & (1 << 4)) ? true : false;
        gp->steel_battalion_in_report.dButtons.Ignition     = (w0 & (1 << 5)) ? true : false;
        gp->steel_battalion_in_report.dButtons.Start        = (w0 & (1 << 6)) ? true : false;
        gp->steel_battalion_in_report.dButtons.MultiMonitorOpenClose   = (w0 & (1 << 7)) ? true : false;

        gp->steel_battalion_in_report.dButtons.MultiMonitorSubMonitor  = (w0 & (1 << 10)) ? true : false;
        gp->steel_battalion_in_report.dButtons.MainMonitorZoomIn       = (w0 & (1 << 11)) ? true : false;
        gp->steel_battalion_in_report.dButtons.MainMonitorZoomOut      = (w0 & (1 << 12)) ? true : false;
        gp->steel_battalion_in_report.dButtons.ForecastShootingSystem = (w0 & (1 << 13)) ? true : false;
        gp->steel_battalion_in_report.dButtons.Manipulator          = (w0 & (1 << 14)) ? true : false;
        gp->steel_battalion_in_report.dButtons.LineColorChange      = (w0 & (1 << 15)) ? true : false;

        gp->steel_battalion_in_report.dButtons.Washing              = (w1 & (1 << 0)) ? true : false;
        gp->steel_battalion_in_report.dButtons.Extinguisher         = (w1 & (1 << 1)) ? true : false;
        gp->steel_battalion_in_report.dButtons.Chaff                = (w1 & (1 << 2)) ? true : false;
        gp->steel_battalion_in_report.dButtons.TankDetach          = (w1 & (1 << 3)) ? true : false;
        gp->steel_battalion_in_report.dButtons.Override             = (w1 & (1 << 4)) ? true : false;

    else {
